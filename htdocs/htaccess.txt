# ImpressCMS host (docs/symfony-host.md). Required: every PHP request must reach
# index.php (routing level 3). Rename this file to .htaccess, or copy the block
# below into your own .htaccess or virtual host. The installer and the system
# module update write .htaccess from this file when the web root has none.
<IfModule mod_rewrite.c>
	Options +SymLinksIfOwnerMatch
	RewriteEngine On
	# For a site in a subdirectory, for example /cms/:
	#RewriteBase /cms/

	# Never served: the host's staging directory, compiled templates, caches,
	# tests, PHP files in uploads/, hidden files and folders, and directories
	# that ImpressCMS protects with their own .htaccess.
	RewriteRule ^(trust_dist|templates_c|cache|tests|plugins/waiting|libraries/tcpdf/tools|install/modules/protector/trust_path)(/|$) - [R=404,L]
	RewriteRule ^uploads/.*\.php$ - [R=404,L]
	RewriteRule (^|/)\. - [R=404,L]

	RewriteRule ^index\.php$ - [L]

	# Every PHP script runs through the host, which runs the legacy script
	RewriteRule \.php$ index.php [L]

	# Unknown paths go to index.php too (routes of the new application)
	RewriteCond %{REQUEST_FILENAME} !-f
	RewriteCond %{REQUEST_FILENAME} !-d
	RewriteRule ^ index.php [L]
</IfModule>
